        // 1. คำนวณและแสดงผล KPI ทั้ง 4 ตัว
        function calculateAndRenderKPIs() {
            const workoutDays = healthData.filter(d => d.duration > 0).length;
            const totalMinutes = healthData.reduce((sum, d) => sum + d.duration, 0);
            const avgSleep = (healthData.reduce((sum, d) => sum + d.sleep, 0) / healthData.length).toFixed(1);
            const avgWater = (healthData.reduce((sum, d) => sum + d.water, 0) / healthData.length).toFixed(2);

            // อัปเดตตัวเลข
            document.getElementById('kpi-workout-days').textContent = workoutDays;
            document.getElementById('kpi-total-minutes').textContent = totalMinutes;
            document.getElementById('kpi-avg-sleep').textContent = avgSleep;
            document.getElementById('kpi-avg-water').textContent = avgWater;

            // อัปเดตสถานะ KPI 1: วันออกกำลังกาย
            const workoutStatusEl = document.getElementById('kpi-workout-status');
            if (workoutDays >= userProfile.targetWorkoutDays) {
                workoutStatusEl.className = "inline-flex items-center gap-1 font-medium text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-md";
                workoutStatusEl.innerHTML = `<i data-lucide="check-circle-2" class="w-3.5 h-3.5"></i> บรรลุเป้าหมาย (${userProfile.targetWorkoutDays} วัน)`;
            } else {
                workoutStatusEl.className = "inline-flex items-center gap-1 font-medium text-amber-600 bg-amber-50 px-2 py-0.5 rounded-md";
                workoutStatusEl.innerHTML = `<i data-lucide="alert-circle" class="w-3.5 h-3.5"></i> ขาดอีก ${userProfile.targetWorkoutDays - workoutDays} วัน`;
            }

            // อัปเดตสถานะ KPI 2: เวลารวม
            const minutesStatusEl = document.getElementById('kpi-minutes-status');
            if (totalMinutes >= userProfile.targetWeeklyMinutes) {
                minutesStatusEl.className = "inline-flex items-center gap-1 font-medium text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-md";
                minutesStatusEl.innerHTML = `<i data-lucide="trending-up" class="w-3.5 h-3.5"></i> ผ่านเกณฑ์ WHO`;
            } else {
                minutesStatusEl.className = "inline-flex items-center gap-1 font-medium text-amber-600 bg-amber-50 px-2 py-0.5 rounded-md";
                minutesStatusEl.innerHTML = `<i data-lucide="clock" class="w-3.5 h-3.5"></i> ต่ำกว่า 150 นาที`;
            }

            // อัปเดตสถานะ KPI 3: การนอน
            const sleepStatusEl = document.getElementById('kpi-sleep-status');
            if (avgSleep >= userProfile.targetSleepMin && avgSleep <= userProfile.targetSleepMax) {
                sleepStatusEl.className = "inline-flex items-center gap-1 font-medium text-indigo-600 bg-indigo-50 px-2 py-0.5 rounded-md";
                sleepStatusEl.innerHTML = `<i data-lucide="check-circle-2" class="w-3.5 h-3.5"></i> อยู่ในเกณฑ์เหมาะสม`;
            } else {
                sleepStatusEl.className = "inline-flex items-center gap-1 font-medium text-amber-600 bg-amber-50 px-2 py-0.5 rounded-md";
                sleepStatusEl.innerHTML = `<i data-lucide="alert-circle" class="w-3.5 h-3.5"></i> ควรปรับเวลานอน`;
            }

            // อัปเดตสถานะ KPI 4: การดื่มน้ำ
            const waterStatusEl = document.getElementById('kpi-water-status');
            if (avgWater >= userProfile.targetWaterLiters) {
                waterStatusEl.className = "inline-flex items-center gap-1 font-medium text-sky-600 bg-sky-50 px-2 py-0.5 rounded-md";
                waterStatusEl.innerHTML = `<i data-lucide="check-circle-2" class="w-3.5 h-3.5"></i> ดื่มน้ำเพียงพอ`;
            } else {
                const diff = (userProfile.targetWaterLiters - avgWater).toFixed(2);
                waterStatusEl.className = "inline-flex items-center gap-1 font-medium text-amber-600 bg-amber-50 px-2 py-0.5 rounded-md";
                waterStatusEl.innerHTML = `<i data-lucide="alert-circle" class="w-3.5 h-3.5"></i> ควรเพิ่มอีก ~${diff} ลิตร`;
            }
        }

        // 3. สร้างรายการกิจกรรมรายวันในตาราง
        function renderDailyTable() {
            const tbody = document.getElementById('daily-log-tbody');
            tbody.innerHTML = '';

            healthData.forEach((item, index) => {
                const isWorkout = item.duration > 0;
                const isSleepGood = item.sleep >= 7.0;
                const isWaterGood = item.water >= 2.5; // เกณฑ์ใกล้เคียงเป้าหมาย 2.8 ลิตร

                // กำหนดไอคอนกิจกรรมอัตโนมัติตามชื่อกิจกรรม
                let iconName = item.icon || 'activity';
                if (item.activity.includes('วิ่ง') || item.activity.includes('เดิน')) iconName = 'footprints';
                else if (item.activity.includes('ปั่น')) iconName = 'bike';
                else if (item.activity.includes('เวท') || item.activity.includes('ยก')) iconName = 'dumbbell';
                else if (item.activity.includes('ว่าย')) iconName = 'waves';
                else if (item.duration === 0) iconName = 'coffee';

                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-50/90 transition-colors group";

                tr.innerHTML = `
                    <td class="py-3.5 px-5 whitespace-nowrap">
                        <div class="font-semibold text-slate-800 font-sans">${item.date}</div>
                        <div class="text-xs text-slate-400">${item.day}</div>
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        <div class="flex items-center gap-2.5">
                            <span class="w-8 h-8 rounded-lg flex items-center justify-center ${isWorkout ? 'bg-teal-50 text-teal-600' : 'bg-slate-100 text-slate-500'}">
                                <i data-lucide="${iconName}" class="w-4 h-4"></i>
                            </span>
                            <span class="font-medium ${isWorkout ? 'text-slate-800' : 'text-slate-500'}">${item.activity}</span>
                        </div>
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        ${isWorkout 
                            ? `<span class="inline-flex items-center px-2.5 py-1 rounded-lg text-xs font-semibold bg-emerald-50 text-emerald-700 border border-emerald-200/60">${item.duration} นาที</span>`
                            : `<span class="inline-flex items-center px-2.5 py-1 rounded-lg text-xs font-medium bg-slate-100 text-slate-500">วันพัก (0 นาที)</span>`
                        }
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        <div class="flex items-center gap-1.5">
                            <span class="font-semibold ${isSleepGood ? 'text-indigo-600' : 'text-amber-600'}">${item.sleep.toFixed(1)} ชม.</span>
                            ${!isSleepGood ? `<span class="text-[11px] text-amber-600 bg-amber-50 px-1.5 py-0.5 rounded">น้อยกว่า 7 ชม.</span>` : ''}
                        </div>
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        <div class="flex items-center gap-2">
                            <div class="w-16 bg-slate-100 rounded-full h-1.5 overflow-hidden hidden sm:block">
                                <div class="bg-sky-500 h-1.5 rounded-full" style="width: ${Math.min(100, (item.water / userProfile.targetWaterLiters) * 100)}%"></div>
                            </div>
                            <span class="font-semibold ${isWaterGood ? 'text-sky-600' : 'text-slate-700'}">${item.water.toFixed(1)} ลิตร</span>
                        </div>
                    </td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        ${getDailyOverallBadge(item)}
                    </td>
                    <td class="py-3.5 px-5 whitespace-nowrap text-right">
                        <button onclick="openModal(${index})" class="inline-flex items-center gap-1 text-xs font-medium text-slate-500 hover:text-teal-600 bg-white hover:bg-teal-50 border border-slate-200 hover:border-teal-200 px-2.5 py-1.5 rounded-lg transition-all cursor-pointer">
                            <i data-lucide="pencil" class="w-3.5 h-3.5"></i>
                            <span>แก้ไข</span>
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // สร้าง Badge ประเมินภาพรวมรายวัน
        function getDailyOverallBadge(item) {
            if (item.duration >= 30 && item.sleep >= 7.0 && item.water >= 2.2) {
                return `<span class="inline-flex items-center gap-1 text-xs font-medium text-emerald-700 bg-emerald-50 px-2.5 py-1 rounded-full"><span class="w-1.5 h-1.5 rounded-full bg-emerald-500"></span> ยอดเยี่ยม</span>`;
            } else if (item.sleep < 7.0 && item.water < 1.8) {
                return `<span class="inline-flex items-center gap-1 text-xs font-medium text-rose-700 bg-rose-50 px-2.5 py-1 rounded-full"><span class="w-1.5 h-1.5 rounded-full bg-rose-500"></span> ควรพักผ่อนและดื่มน้ำเพิ่ม</span>`;
            } else {
                return `<span class="inline-flex items-center gap-1 text-xs font-medium text-sky-700 bg-sky-50 px-2.5 py-1 rounded-full"><span class="w-1.5 h-1.5 rounded-full bg-sky-500"></span> อยู่ในเกณฑ์ดี</span>`;
            }
        }

        function renderInsights() {
            const workoutDays = healthData.filter(d => d.duration > 0).length;
            const totalMinutes = healthData.reduce((sum, d) => sum + d.duration, 0);
            const avgSleep = (healthData.reduce((sum, d) => sum + d.sleep, 0) / healthData.length).toFixed(1);

            const waterGap = (userProfile.targetWaterLiters - avgWater).toFixed(2);

            const insightsList = document.getElementById('health-insights-list');
            insightsList.innerHTML = `
                <li class="flex items-start gap-2.5">
                    <span class="w-5 h-5 rounded-full bg-emerald-100 text-emerald-600 flex items-center justify-center shrink-0 mt-0.5 text-xs font-bold">1</span>
                    <span><strong class="text-slate-800">การออกกำลังกาย:</strong> ทำได้ <strong class="text-emerald-600">${workoutDays} วัน (${totalMinutes} นาที)</strong> ${workoutDays >= 5 ? 'บรรลุเป้าหมาย 5 วัน/สัปดาห์ และเกินเกณฑ์ขั้นต่ำ 150 นาทีของ WHO ได้อย่างยอดเยี่ยม' : 'ควรเพิ่มวันออกกำลังกายให้ครบ 5 วันตามเป้าหมาย'}</span>
                </li>
                <li class="flex items-start gap-2.5">
                    <span class="w-5 h-5 rounded-full bg-indigo-100 text-indigo-600 flex items-center justify-center shrink-0 mt-0.5 text-xs font-bold">2</span>
                    <span><strong class="text-slate-800">คุณภาพการนอน:</strong> เฉลี่ย <strong class="text-indigo-600">${avgSleep} ชม./วัน</strong> อยู่ในเกณฑ์มาตรฐาน (7-9 ชม.) มีเพียงวันที่ 8 ก.ย. (6.5 ชม.) และ 10 ก.ย. (6.8 ชม.) ที่นอนน้อยกว่าเกณฑ์เล็กน้อย</span>
                </li>
                <li class="flex items-start gap-2.5">
                    <span class="w-5 h-5 rounded-full bg-sky-100 text-sky-600 flex items-center justify-center shrink-0 mt-0.5 text-xs font-bold">3</span>
                    <span><strong class="text-slate-800">การดื่มน้ำ (จุดที่ควรเพิ่ม):</strong> สำหรับน้ำหนัก 85 กก. ควรดื่มน้ำ ~2.8 ลิตร/วัน แต่ค่าเฉลี่ยอยู่ที่ <strong class="text-amber-600">${avgWater} ลิตร/วัน</strong> ${waterGap > 0 ? `(ควรดื่มเพิ่มอีกวันละประมาณ ${waterGap} ลิตร หรือ 3-4 แก้ว)` : '(ดื่มน้ำได้ตามเป้าหมายแล้ว)'}</span>
                </li>
            `;
        }

        // ฟังก์ชันจัดการ Modal แก้ไขข้อมูล
        function openModal(dayIndex = 0) {
            const modal = document.getElementById('editModal');
            const select = document.getElementById('daySelect');
            
            // เติมตัวเลือกวันที่
            select.innerHTML = healthData.map((d, idx) => 
                `<option value="${idx}" ${idx === dayIndex ? 'selected' : ''}>${d.date} (${d.day}) - ${d.activity}</option>`
            ).join('');

            populateModalFields();
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeModal() {
            const modal = document.getElementById('editModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        function populateModalFields() {
            const idx = document.getElementById('daySelect').value;
            const item = healthData[idx];
            document.getElementById('activityInput').value = item.activity;
            document.getElementById('durationInput').value = item.duration;
            document.getElementById('sleepInput').value = item.sleep;
            document.getElementById('waterInput').value = item.water;
        }

        function saveDayData(event) {
            event.preventDefault();
            const idx = parseInt(document.getElementById('daySelect').value, 10);
            const activity = document.getElementById('activityInput').value.trim();
            const duration = parseInt(document.getElementById('durationInput').value, 10) || 0;
            const sleep = parseFloat(document.getElementById('sleepInput').value) || 0;
            const water = parseFloat(document.getElementById('waterInput').value) || 0;

            healthData[idx] = {
                ...healthData[idx],
                activity: duration === 0 && activity === '' ? 'พักผ่อน' : activity,
                duration: duration,
                sleep: sleep,
                water: water
            };

            closeModal();
            updateDashboard();
            showToast(`อัปเดตข้อมูลวันที่ ${healthData[idx].date} เรียบร้อยแล้ว`);
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-message').textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }

        // เริ่มต้นทำงานเมื่อโหลดหน้าเว็บเสร็จ
        window.addEventListener('DOMContentLoaded', () => {
            updateDashboard();
        });
    </script>
